#!/bin/bash

# Exit immediately if a command fails
set -e

echo "================================================================="
echo "STARTING TOTAL SPATIAL METABOLOMICS PIPELINE"
echo "================================================================="
echo "Timestamp: $(date)"

# Define Paths relative to the execution folder
BASE_DIR="/mnt/c/Users/great/Downloads/Test_Example"
SCRIPT_DIR="$BASE_DIR/bash_scripts"

# Define Data Input Files (Now looking in the root Test_Example folder)
IMZML_FILE="$BASE_DIR/3262-3_1_dmamousekidney_20260518_hv.imzML"

# Intermediate and Output Suffix Paths
INTX_ZST="$BASE_DIR/3262-3_1_dmamousekidney_20260518_hv_intx.tsv.zst"
FILTERED_TSV="$BASE_DIR/3262-3_1_dmamousekidney_20260518_hv_intx.tsv_filtered.tsv"
FEATURE_TREE="$BASE_DIR/3262-3_1_dmamousekidney_20260518_hv_feature_tree.tre"
PIXEL_TREE="$BASE_DIR/3262-3_1_dmamousekidney_20260518_hv_pixel_tree.tre"
OUTPUT_TOKEN="$BASE_DIR/hcdist_output"

# Configuration Parameters
CORES=4
METABOLITE_CLUSTERS=10
PIXEL_CLUSTERS=6

echo "Moving into base directory..."
cd "$BASE_DIR"

echo "Cleaning scripts of hidden Windows line endings..."
find "$SCRIPT_DIR" -name "go_*.sh" -exec dos2unix {} +

# -----------------------------------------------------------------
# STAGE 0: Cardinal Matrix Preprocessing
# -----------------------------------------------------------------
echo -e "\n STEP 0: Running Cardinal Extraction (Parsing .imzML/.ibd)..."
cd "$SCRIPT_DIR"

# Dynamically generate the required temporary parameter list file
CARDINAL_PARAM="hcdist_imzml_param_list.txt"
echo "${IMZML_FILE} ${CORES}" > "$CARDINAL_PARAM"

bash go_cardinal_all.sh "$CARDINAL_PARAM"
rm -f "$CARDINAL_PARAM" 

# Verify Step 0 success
if [ ! -f "$INTX_ZST" ]; then
    echo " Error: Cardinal extraction failed. ${INTX_ZST} was not created."
    exit 1
fi
echo "Step 0 Complete! Compressed intensity matrix generated."

# -----------------------------------------------------------------
# STAGE 1: Data Filtering
# -----------------------------------------------------------------
echo -e "\n STEP 1: Running Data Filtering Strategy..."
cd "$SCRIPT_DIR"

# Create the temporary file list that go_filter_data.sh expects
FILTER_PARAM="hcdist_filter_param_list.txt"
echo "${INTX_ZST}" > "$FILTER_PARAM"

# Run the script passing the list file, NOT the binary file
bash go_filter_data.sh "$FILTER_PARAM"
rm -f "$FILTER_PARAM"

# Verify Step 1 success
if [ ! -f "$FILTERED_TSV" ]; then
    echo " Error: Filtered file was not created successfully."
    exit 1
fi
echo "Step 1 Complete! Filtered data file generated."
#-----------------------------------------------------------------
# STAGE 2: Feature Clustering Tree
# -----------------------------------------------------------------
echo -e "\n STEP 2: Generating Metabolite Relationship Tree..."
bash go_cluster_features.sh "$FILTERED_TSV" "$FEATURE_TREE" $CORES
echo "Step 2 Complete! Feature Tree (.tre) generated."

# -----------------------------------------------------------------
# STAGE 3: Pixel Clustering Tree (Spatial Domains)
# -----------------------------------------------------------------
echo -e "\n STEP 3: Detecting Spatial Domains (Pixel Tree)..."
bash go_detect_spatial_domains.sh "$FILTERED_TSV" "$PIXEL_TREE" $CORES
echo "Step 3 Complete! Pixel Tree (.tre) generated."

# -----------------------------------------------------------------
# STAGE 4: Generate Map Artifacts & Clustered Heatmap
# -----------------------------------------------------------------
echo -e "\n STEP 4: Assembling Final Clustered Result Matrix..."
bash go_generate_hcdist_result_files.sh \
  "$FILTERED_TSV" \
  "$FEATURE_TREE" \
  "$PIXEL_TREE" \
  "$OUTPUT_TOKEN" \
  $METABOLITE_CLUSTERS \
  $PIXEL_CLUSTERS

echo -e "\n================================================================="
echo "ALL STAGES COMPLETE!"
echo "================================================================="
echo "Your final visual heatmap is waiting here:"
echo "${OUTPUT_TOKEN}_heatmap.jpg"
echo "Timestamp: $(date)"
