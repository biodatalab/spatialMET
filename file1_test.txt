# Use the official R base image
FROM r-base:latest

# Install system dependencies (needed for R packages and hcdist)
RUN apt-get update -qq && apt-get -y --no-install-recommends install \
    libfontconfig1-dev \
    libcurl4-openssl-dev \
    gcc \
    g++ \
    libfreetype6-dev \
    libglib2.0-dev \
    libcairo2-dev \
    libharfbuzz-dev \
    libfribidi-dev \
    libtiff5-dev \
    libxml2-dev \
    libudunits2-dev \
    libgdal-dev \
    libtool \
    zstd \
    make \
    && rm -rf /var/lib/apt/lists/*

# Create the app directory
RUN mkdir -p /app

# Copy the Shiny app and hcdist pipeline into the container
COPY shinyapp_stage2 /app
COPY hcdist_stage1 /app/hcdist_stage1

# Set working directory to the app
WORKDIR /app

# Make hcdist binaries executable
RUN chmod +x /app/hcdist_stage1/bash_scripts/*.sh && \
    chmod +x /app/hcdist_stage1/hcdist/hcdist && \
    chmod +x /app/hcdist_stage1/hcdist/maldi_image_from_clusters && \
    # Create symlink so scripts find the binaries
    ln -s /app/hcdist_stage1/hcdist /app/hcdist_stage1/maldi-clustering-and-imaging-main

# Install all required R packages
RUN R -e "install.packages(c('httpuv', 'shiny', 'shinyDashboardThemeDIY', 'dashboardthemes', 'shinydashboard', 'shinyvalidate', 'DT', 'ggiraph', 'tidyverse', 'markdown', 'ggnewscale', 'khroma', 'units', 'sf', 'spdep', 'sfsmisc', 'plotly', 'colourpicker', 'shinyjs', 'vroom', 'fastcluster', 'future', 'promises', 'processx'))" && \
    R -e "if (!require('BiocManager', quietly = TRUE)) install.packages('BiocManager'); BiocManager::install(c('Cardinal', 'BiocParallel', 'S4Vectors', 'DelayedArray', 'Matrix', 'limma', 'splines', 'uwot', 'spatstat.geom', 'spatstat.explore', 'mxfda', 'WGCNA', 'irlba', 'visNetwork', 'igraph', 'SummarizedExperiment', 'BiocGenerics', 'ProtGenerics', 'data.table', 'tibble', 'magrittr'))"

# Expose port 3838 (Shiny default)
EXPOSE 3838

# Run the Shiny app on container start
CMD ["R", "-e", "shiny::runApp('.', host = '0.0.0.0', port = 3838)"]
